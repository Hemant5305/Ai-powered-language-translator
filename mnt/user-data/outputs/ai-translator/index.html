HawkEye AI is a web technology project exploring how modern AI APIs can transform everyday communication tools. Using the Anthropic Claude API for intelligent translation, the Web Speech API for voice interaction, and vanilla JavaScript for state management — no heavy frameworks, just clean code.

const translate = async (text, target) => {
  const response = await fetch(API_URL, {
    method: 'POST',
    body: JSON.stringify({
      model: 'claude-sonnet-4-6',
      messages: [{
        role: 'user',
        content: buildPrompt(text, target)
      }]
    })
  });
  return response.json();
};
